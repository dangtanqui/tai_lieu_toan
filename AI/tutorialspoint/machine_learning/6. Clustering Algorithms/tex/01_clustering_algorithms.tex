%==============================================================================
\section{Thuật toán phân cụm (Clustering Algorithms)}
\label{sec:clustering-algorithms}
%==============================================================================

\begin{dinhnghia}[title={Thuật toán phân cụm}]
Là một trong những phương pháp \textbf{học không giám sát} hữu ích nhất: tìm độ tương tự và quy luật quan hệ giữa các mẫu dữ liệu, rồi gom các mẫu tương tự (theo feature) thành cụm.
\end{dinhnghia}

\begin{ynghia}[title={Tầm quan trọng}]
Phân cụm xác định nhóm nội tại trong dữ liệu \textbf{không nhãn}.
Mỗi giả định về độ tương tự giữa điểm dữ liệu tạo ra các cụm khác nhau nhưng đều có thể hợp lệ.
\end{ynghia}

\subsection{Phương pháp hình thành cụm}

\begin{phuongphap}[title={Không nhất thiết hình cầu}]
Cụm không bắt buộc có dạng hình cầu. Các hướng hình thành cụm phổ biến:
\begin{itemize}
  \item \textbf{Dựa trên mật độ (Density-based)}: Cụm là vùng \textbf{mật độ cao}; ưu điểm là độ chính xác cao, có thể gộp hai cụm.
  \item \textbf{Phân cấp (Hierarchical-based)}: Cụm dạng \textbf{cây} theo thứ bậc. Cụm kết hợp - Agglomerative (gộp từ dưới lên), cụm phân chia - Divisive (chia từ trên xuống).
  \item \textbf{Phân hoạch (Partitioning)}: Chia đối tượng thành $k$ cụm; số cụm bằng số phân hoạch.
  \item \textbf{Lưới (Grid)}: Cụm dạng \textbf{lưới}; ưu điểm là thao tác trên lưới nhanh, độc lập số đối tượng.
\end{itemize}
\end{phuongphap}

\subsection{Các thuật toán phân cụm chính}

\begin{phuongphap}[title={Danh sách}]
  \begin{itemize}
    \item \textbf{K-Means}: Tính centroid và lặp đến centroid tối ưu; giả định \textbf{đã biết} số cụm $K$ (flat clustering).
    \item \textbf{K-Medoids}: Cải tiến K-means: chọn $k$ medoid ngẫu nhiên, gán điểm, cập nhật medoid (tối thiểu tổng khoảng cách trong cụm), lặp đến hội tụ.
    \item \textbf{Mean-Shift}: Phân cụm mạnh, \textbf{không tham số hóa} giả định phân phối (non-parametric).
    \item \textbf{DBSCAN}: Dựa trên mật độ phổ biến nhất. Cần hai tham số: \texttt{minPts} (số láng giềng tối thiểu) và \texttt{eps} (khoảng cách tối đa giữa điểm lõi).
    \item \textbf{OPTICS}: Giống DBSCAN nhưng xử lý cụm mật độ khác nhau, nhiễu, và cho cấu trúc phân cấp.
    \item \textbf{HDBSCAN}: Mở rộng DBSCAN: cụm mật độ thay đổi, hình dạng/kích thước đa dạng.
    \item \textbf{BIRCH}: Phân cấp cho dataset lớn; cây cụm con bằng chia đệ quy đến khi đạt tiêu chí dừng.
    \item \textbf{Lan truyền liên kết (Affinity Propagation)}: Xác định ``mẫu đại diện'' (exemplar); không cần chỉ định trước số cụm --- hữu ích khám phá dữ liệu.
    \item \textbf{Phân cấp kết hợp (Agglomerative Clustering)}: Phân cấp từ dưới lên: mỗi điểm là một cụm, gộp cụm gần nhất; sinh \textbf{cây phân cấp} (dendrogram).
    \item \textbf{Mô hình hỗn hợp Gaussian (Gaussian Mixture Model)}: Giả định dữ liệu sinh từ hỗn hợp phân phối Gaussian; mỗi Gaussian là một cụm.
  \end{itemize}
\end{phuongphap}

\subsection{Đo hiệu năng phân cụm}

\begin{dongco}[title={Bối cảnh}]
Với học có giám sát, đánh giá mô hình tương đối dễ vì có nhãn.
Với học không giám sát, dữ liệu không nhãn nhưng vẫn có metric so sánh mô hình với nhau (không chứng minh ``đúng tuyệt đối'').
\end{dongco}

\subsubsection{Phân tích Silhouette (Silhouette Analysis)}

\begin{tinhchat}[title={Silhouette score}]
Đo khoảng cách giữa các cụm; hỗ trợ chọn số cụm qua điểm Silhouette.
Miền giá trị $[-1, 1]$:
\begin{itemize}
  \item Gần $+1$: mẫu xa cụm láng giềng.
  \item $0$: mẫu gần ranh giới giữa hai cụm.
  \item Gần $-1$: mẫu có thể gán nhầm cụm.
\end{itemize}
\end{tinhchat}

\begin{phuongphap}[title={Công thức}]
Với mỗi mẫu: $s = \dfrac{b - a}{\max(a, b)}$, trong đó $a$ = khoảng cách trung bình trong cụm, $b$ = khoảng cách trung bình đến cụm láng giềng gần nhất.
\end{phuongphap}

\subsubsection{Davies--Bouldin Index}

\begin{tinhchat}[title={DB index}]
Đánh giá cụm có \textbf{cách xa} nhau không, cụm có \textbf{dày} không; DB index \textbf{càng nhỏ} càng tốt.
\[
\mathrm{DB} = \frac{1}{n}\sum_{i=1}^{n} \max_{j \neq i}
\left( \frac{\sigma_i + \sigma_j}{d(c_i, c_j)} \right)
\]
$n$: số cụm; $\sigma_i$: khoảng cách trung bình các điểm trong cụm đến centroid $c_i$.
\end{tinhchat}

\subsubsection{Dunn Index}

\begin{tinhchat}[title={Dunn Index}]
Tương tự DB index nhưng:
\begin{itemize}
  \item Dunn chỉ xét \textbf{trường hợp xấu nhất} (cụm gần nhau nhất).
  \item DB xét \textbf{độ phân tán} (dispersion) và \textbf{độ tách biệt} (separation) của \textbf{mọi} cặp cụm.
  \item Dunn \textbf{tăng} khi hiệu năng tốt; DB tốt khi cụm tách xa và dày.
\end{itemize}
\end{tinhchat}

\subsection{Ứng dụng phân cụm}

\begin{ungdung}[title={Ứng dụng phân cụm}]
  \begin{itemize}
    \item Tóm tắt và nén dữ liệu: xử lý ảnh, lượng tử hóa vector.
    \item Gom sản phẩm hoặc người dùng tương tự: hệ thống gợi ý và phân khúc khách hàng.
    \item Bước trung gian khai phá dữ liệu: tóm tắt cho phân loại, kiểm định, sinh giả thuyết.
    \item Gom cụm xu hướng tương tự theo thời gian: phát hiện xu hướng trong dữ liệu động.
    \item Phân tích mạng xã hội và dữ liệu sinh học: chuỗi trong ảnh/video/âm thanh; phân cụm trong phân tích dữ liệu sinh học.
  \end{itemize}
\end{ungdung}
