\subsection{Học máy là gì?}

\begin{ynghia}[title={Định nghĩa}]
Xem Mục~\ref{sec:ml-la-gi}. ML dùng dữ liệu và thuật toán để học pattern ẩn và dự đoán trên dữ liệu mới.
\end{ynghia}

\subsection{Các loại học máy}

\begin{ynghia}[title={Ba--bốn nhóm (theo nguồn)}]
Nguồn liệt kê supervised, unsupervised, reinforcement; semi-supervised được bổ sung ở Mục~\ref{sec:bon-kieu-hoc}. Chi tiết thuật toán từng nhóm: Mục~\ref{sec:mo-hinh-ml} trở đi.
\end{ynghia}

\begin{ynghia}[title={Ba nhóm bài toán (theo nguồn)}]
Clustering, Association, Dimensionality reduction.
\end{ynghia}

\begin{phuongphap}[title={Một số thuật toán unsupervised (theo nguồn)}]
K-Means, PCA, Hierarchical Clustering, DBSCAN, Agglomerative Clustering, Apriori, Autoencoder, RBM.
\end{phuongphap}

\begin{ynghia}[title={Reinforcement learning}]
Định nghĩa và thuật toán RL: Mục~\ref{sec:reinforcement-learning}.
\end{ynghia}

\subsection{Use cases theo loại học}

\begin{phuongphap}[title={Supervised learning}]
Image classification, spam filtering, dự đoán giá nhà, nhận dạng chữ ký, dự báo thời tiết, dự đoán giá cổ phiếu.
\end{phuongphap}

\begin{phuongphap}[title={Unsupervised learning}]
Anomaly detection, recommendation systems, customer segmentation, fraud detection, NLP, genetic search.
\end{phuongphap}

\begin{phuongphap}[title={Reinforcement learning}]
Autonomous vehicles, robotics, game playing.
\end{phuongphap}

\subsection{Điều kiện tiên quyết để bắt đầu}

\begin{dongco}[title={Học ML dễ hơn khi chuẩn bị đúng}]
Để bắt đầu học ML, bạn nên nắm một số nền tảng về khoa học máy tính, đồng thời quen với lập trình, thư viện và toán--thống kê.
\end{dongco}

\subsubsection{Ngôn ngữ lập trình: Python hoặc R}

\begin{phuongphap}[title={Gợi ý (theo nguồn)}]
Có nhiều ngôn ngữ dùng được (C++, Java, Python, R, Julia, \ldots), nhưng Python rất phổ biến trong ML và Data Science.
Trong tutorial này, nguồn nói sẽ dùng Python và/hoặc R.
\end{phuongphap}

\begin{phuongphap}[title={Các chủ đề Python cơ bản nên nắm}]
Variables, basic data types; data structures (list, set, dictionaries); loops and conditionals; functions; string formatting; classes and objects.
\end{phuongphap}
