%==============================================================================
\section{Hồi quy đa thức (Polynomial Regression)}
\label{sec:polynomial-regression}
%==============================================================================

\begin{dinhnghia}[title={Hồi quy đa thức}]
Polynomial Linear Regression mô hình hóa quan hệ giữa biến độc lập và phụ thuộc bằng \textbf{hàm đa thức bậc $n$}, vượt qua giới hạn tuyến tính của hồi quy đơn và hồi quy tuyến tính đa biến thuần túy.
\end{dinhnghia}

\subsection{Vì sao cần hồi quy đa thức?}

\begin{dongco}[title={Chọn linear hay polynomial}]
Trong ML và data science, chọn hồi quy tuyến tính hay đa thức phụ thuộc \textbf{đặc tính dataset}.
Dữ liệu \textbf{phi tuyến} không khớp bằng hồi quy tuyến tính; nếu ép tuyến tính, mô hình không bắt được pattern cong.
\end{dongco}

\tsimg{05_polynomial_regression/img01.jpg}

\begin{ynghia}[title={Đọc hình so sánh}]
Mô hình tuyến tính đơn khó khớp các điểm; mô hình đa thức khớp phần lớn điểm tốt hơn.
\end{ynghia}

\subsection{Phương trình mô hình}

\begin{dinhnghia}[title={Đa thức bậc $n$}]
\[
y = w_0 + w_1 x + w_2 x^2 + w_3 x^3 + \cdots + w_n x^n + \epsilon
\]
\begin{itemize}
  \item $y$: biến phụ thuộc (output);
  \item $x$: biến độc lập (input);
  \item $w_0, w_1, \ldots, w_n$: hệ số;
  \item $n$: bậc đa thức (lũy thừa cao nhất của $x$);
  \item $\epsilon$: sai số / phần dư.
\end{itemize}
\end{dinhnghia}

\begin{dinhnghia}[title={Đa thức bậc 2 (parabol)}]
\[
y = w_0 + w_1 x + w_2 x^2 + \epsilon
\]
Khớp đường cong parabol với các điểm dữ liệu.
\end{dinhnghia}

\subsection{Cách hoạt động}

\begin{phuongphap}[title={Cơ chế trong học máy}]
Hồi quy đa thức hoạt động tương tự hồi quy tuyến tính nhưng được mô hình như \textbf{hồi quy tuyến tính đa biến}:
biến đổi feature gốc thành $x, x^2, \ldots, x^n$; các cột này được coi là feature độc lập riêng; huấn luyện \texttt{LinearRegression} trên không gian mở rộng.
\end{phuongphap}

\begin{luuy}[title={Khác MLR thuần}]
MLR giả định feature gốc tuyến tính với target; trong hồi quy đa thức, các cột $x^k$ \textbf{phụ thuộc} cùng một $x$ ban đầu.
\end{luuy}

\subsection{Triển khai bằng Python (Scikit-learn)}

\begin{phuongphap}[title={Thư viện}]
Triển khai bằng Python; dùng Scikit-learn để xây mô hình hồi quy.
\end{phuongphap}

\begin{vidu}[title={Bước 1: Chuẩn bị dữ liệu --- import}]
Chuẩn bị dữ liệu là bước quan trọng. Dùng pandas đọc CSV, NumPy cho mảng, \texttt{PolynomialFeatures} để biến đổi bậc:

\begin{lstlisting}
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import PolynomialFeatures
\end{lstlisting}
\end{vidu}

\begin{vidu}[title={Nạp dataset}]
\begin{lstlisting}
data = pd.read_csv('data/ice_cream_selling_data.csv')
print(data.head())
\end{lstlisting}

\textbf{Output \texttt{head()}:}
\begin{verbatim}
   Temperature  IceCreamSales
0    -4.662263      41.842986
1    -4.316559      34.661120
...
\end{verbatim}
\end{vidu}

\begin{vidu}[title={Tách $X$ và $y$}]
\begin{lstlisting}
X = data.iloc[:, 0].values.reshape(-1, 1)
y = data.iloc[:, 1].values
\end{lstlisting}
\end{vidu}

\begin{vidu}[title={Trực quan dữ liệu gốc}]
\begin{lstlisting}
plt.scatter(X, y, color="green")
plt.title("Original Data")
plt.xlabel("Temperature (C)")
plt.ylabel("Ice Cream Sales (units)")
plt.show()
\end{lstlisting}

\tsimg{05_polynomial_regression/img02.jpg}

\begin{ynghia}[title={Nhận xét đồ thị}]
Dữ liệu gần dạng \textbf{parabol} (đa thức bậc 2) $\Rightarrow$ mô hình hóa bằng hồi quy đa thức bậc 2 phù hợp quan hệ nhiệt độ -- doanh số kem.
\end{ynghia}
\end{vidu}

\begin{vidu}[title={Tạo đặc trưng đa thức bậc 2}]
\begin{lstlisting}
degree = 2
poly_features = PolynomialFeatures(degree=degree)
X_poly = poly_features.fit_transform(X)
\end{lstlisting}

\texttt{X\_poly}: feature đa thức từ $X$; kích thước $(49, 3)$ (cột hằng, $x$, $x^2$).
\end{vidu}

\begin{vidu}[title={Bước 2: Huấn luyện mô hình}]
Hồi quy đa thức là trường hợp đặc biệt của hồi quy tuyến tính:

\begin{lstlisting}
from sklearn.linear_model import LinearRegression
lr_model = LinearRegression()
lr_model.fit(X_poly, y)
\end{lstlisting}
\end{vidu}

\begin{vidu}[title={Bước 3: Dự đoán và kiểm tra}]
Dự đoán trên dữ liệu hiện có trước khi dự đoán điểm mới:

\begin{lstlisting}
y_pred = lr_model.predict(X_poly)
df = pd.DataFrame({'Actual Values': y, 'Predicted Values': y_pred})
print(df)
\end{lstlisting}

\textbf{Output trích (5 dòng đầu bài gốc):}
\begin{verbatim}
   Actual Values  Predicted Values
0       41.842986         46.564507
1       34.661120         40.600548
2       39.383001         38.915089
3       37.539845         34.749272
4       32.284531         29.331940
\end{verbatim}
So sánh cột thực tế và dự đoán.
\end{vidu}

\begin{vidu}[title={Bước 4: Đánh giá hiệu năng --- $R^2$}]
\begin{lstlisting}
from sklearn.metrics import r2_score
r2 = r2_score(y, y_pred)
print(r2)
\end{lstlisting}

\textbf{Output:} \texttt{0.9321137090423877}

\begin{ynghia}[title={Giải thích $R^2$}]
$R^2$ đo tỷ lệ phương sai biến phụ thuộc có thể dự đoán từ biến độc lập.

Điểm cao $\Rightarrow$ khớp tốt hơn; 1 = khớp hoàn hảo; 0 = không có quan hệ tuyến tính giữa dự đoán và thực tế.

Với $R^2 \approx 0{,}932$: khoảng 93\% biến thiên output được giải thích bởi input; phần lớn điểm nằm quanh đường cong hồi quy.
\end{ynghia}
\end{vidu}

\begin{vidu}[title={Bước 5: Trực quan kết quả hồi quy đa thức}]
\begin{lstlisting}
plt.scatter(X, y, color="green")
plt.plot(X, y_pred, color='red',
         label=f'Polynomial Regression (degree={degree})')
plt.xlabel("Temperature (C)")
plt.ylabel("Ice Cream Sales (units)")
plt.legend()
plt.title('Polynomial Regression')
plt.show()
\end{lstlisting}

\begin{ynghia}[title={Đọc biểu đồ}]
Đường đỏ (parabol bậc 2) khớp tốt dữ liệu gốc; dùng để dự đoán; giá trị dự đoán gần giá trị thực.
\end{ynghia}
\end{vidu}

\begin{vidu}[title={Bước 6: Dự đoán dữ liệu mới}]
Nhiệt độ $1{,}9929\,^{\circ}\mathrm{C}$:

\begin{lstlisting}
X_new = np.array([[1.9929]])
X_new_poly = poly_features.transform(X_new)
y_new_pred = lr_model.predict(X_new_poly)
print(y_new_pred)
\end{lstlisting}

\textbf{Output:} \texttt{[8.57450466]} --- dự đoán khoảng 8.57 đơn vị kem bán ra.
\end{vidu}

\begin{luuy}[title={Bậc đa thức và overfitting}]
Bậc quá cao dễ \textbf{overfitting}; khi triển khai thực tế nên thử cross-validation hoặc regularization.
\end{luuy}
