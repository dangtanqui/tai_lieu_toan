\newcommand{\mlcbframegrid}[4]{%
  \mlcbframecols=#1\relax
  \mlcbframeend=#4\relax
  \setlength{\mlcbframew}{\dimexpr(0.97\linewidth)/\mlcbframecols\relax}%
  \par\smallskip\begin{center}%
  \mlcbframeidx=#3\relax
  \mlcbframecol=0
  \loop\ifnum\mlcbframeidx>\mlcbframeend\else
    \begin{minipage}[t]{\mlcbframew}%
      \centering
      \edef\mlcbframename{#2-\the\mlcbframeidx}%
      \IfFileExists{images/\mlcbframename.png}{%
        \includegraphics[width=\linewidth]{\mlcbframename}%
      }{%
        \fbox{\parbox[c][1cm][c]{\dimexpr\linewidth-2\fboxsep\relax}{\centering\scriptsize\mlcbframename}}%
      }%
    \end{minipage}%
    \advance\mlcbframecol by 1
    \ifnum\mlcbframecol=\mlcbframecols\relax
      \par\medskip
      \mlcbframecol=0
    \else
      \hspace{0.25em}%
    \fi
    \advance\mlcbframeidx by 1
  \repeat
  \end{center}\smallskip
}
