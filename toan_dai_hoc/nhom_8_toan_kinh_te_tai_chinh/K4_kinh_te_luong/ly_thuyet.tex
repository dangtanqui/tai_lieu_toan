\section{Mô hình hồi quy tuyến tính}

\begin{dinhnghia}[title={Mô hình hồi quy đơn}]
  $Y_i = \beta_0 + \beta_1 X_i + \varepsilon_i$, $i = 1, \ldots, n$. Giả thiết: $E(\varepsilon_i) = 0$, $\text{Var}(\varepsilon_i) = \sigma^2$, $\text{Cov}(\varepsilon_i, \varepsilon_j) = 0$.
\end{dinhnghia}

\begin{phuongphap}[title={OLS --- Bình phương tối thiểu thông thường}]
  $\min \sum_{i=1}^{n}(Y_i - \beta_0 - \beta_1 X_i)^2$.

  $\hat{\beta}_1 = \dfrac{\sum(X_i - \bar{X})(Y_i - \bar{Y})}{\sum(X_i - \bar{X})^2}$, $\hat{\beta}_0 = \bar{Y} - \hat{\beta}_1\bar{X}$.
\end{phuongphap}

\begin{dinhnghia}[title={Mô hình hồi quy bội}]
  $\mathbf{Y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}$. OLS: $\hat{\boldsymbol{\beta}} = (\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T\mathbf{Y}$.
\end{dinhnghia}

\begin{dinhly}[title={Gauss--Markov}]
  Dưới giả thiết cổ điển (sai số trung bình 0, đồng phương sai, không tương quan), $\hat{\boldsymbol{\beta}}_{\text{OLS}}$ là ước lượng tuyến tính không chệch tốt nhất (BLUE).
\end{dinhly}

\begin{congthuc}[title={$R^2$ và $R^2$ hiệu chỉnh}]
  $R^2 = 1 - \dfrac{SSR}{SST}$, $R^2_{\text{adj}} = 1 - \dfrac{SSR/(n-k-1)}{SST/(n-1)}$.
\end{congthuc}

\section{Kiểm định và chẩn đoán}

\begin{phuongphap}[title={Kiểm định ý nghĩa hệ số}]
  $t = \dfrac{\hat{\beta}_j}{se(\hat{\beta}_j)} \sim t_{n-k-1}$ dưới $H_0: \beta_j = 0$.

  Kiểm định $F$ tổng thể: $F = \dfrac{R^2/k}{(1-R^2)/(n-k-1)}$.
\end{phuongphap}

\begin{dinhnghia}[title={Đa cộng tuyến}]
  Các biến độc lập tương quan cao. Hệ quả: $\text{Var}(\hat{\beta}_j)$ lớn. Phát hiện: VIF $= \dfrac{1}{1 - R_j^2} > 10$.

  Khắc phục: loại biến, Ridge regression, PCA.
\end{dinhnghia}

\begin{dinhnghia}[title={Phương sai thay đổi (Heteroscedasticity)}]
  $\text{Var}(\varepsilon_i) = \sigma_i^2$ thay đổi. Kiểm định: Breusch--Pagan, White.

  Khắc phục: WLS, robust standard errors (HC).
\end{dinhnghia}

\begin{dinhnghia}[title={Tự tương quan}]
  $\text{Cov}(\varepsilon_t, \varepsilon_s) \ne 0$. Kiểm định: Durbin--Watson, Breusch--Godfrey.

  Khắc phục: GLS, Newey--West standard errors.
\end{dinhnghia}

\section{Mô hình mở rộng}

\begin{dinhnghia}[title={Biến giả (Dummy)}]
  $D_i \in \{0, 1\}$: mã hóa biến định tính. Bẫy biến giả: $k$ nhóm dùng $k - 1$ biến giả.
\end{dinhnghia}

\begin{dinhnghia}[title={Biến công cụ (IV) và 2SLS}]
  Biến nội sinh: $\text{Cov}(X, \varepsilon) \ne 0$. Dùng công cụ $Z$: $\text{Cov}(Z, X) \ne 0$, $\text{Cov}(Z, \varepsilon) = 0$.

  2SLS: (1) hồi quy $X$ theo $Z$, (2) hồi quy $Y$ theo $\hat{X}$.
\end{dinhnghia}

\section{Chuỗi thời gian}

\begin{dinhnghia}[title={Chuỗi dừng}]
  $\{Y_t\}$ dừng (yếu): $E(Y_t)$, $\text{Var}(Y_t)$ không đổi, $\text{Cov}(Y_t, Y_{t+k})$ chỉ phụ thuộc $k$.
\end{dinhnghia}

\begin{dinhnghia}[title={Mô hình ARMA}]
  $\text{AR}(p)$: $Y_t = c + \phi_1 Y_{t-1} + \cdots + \phi_p Y_{t-p} + \varepsilon_t$.

  $\text{MA}(q)$: $Y_t = \mu + \varepsilon_t + \theta_1\varepsilon_{t-1} + \cdots + \theta_q\varepsilon_{t-q}$.

  $\text{ARMA}(p,q)$: kết hợp AR và MA.
\end{dinhnghia}

\begin{dinhnghia}[title={Nghiệm đơn vị và ARIMA}]
  Kiểm định ADF: $H_0$: có nghiệm đơn vị (chuỗi không dừng).

  ARIMA$(p,d,q)$: sai phân $d$ lần trước khi áp ARMA$(p,q)$.
\end{dinhnghia}

\begin{dinhnghia}[title={Đồng liên kết (Cointegration)}]
  Các chuỗi $I(1)$ nhưng tổ hợp tuyến tính dừng $I(0)$. Kiểm định: Engle--Granger, Johansen. Mô hình ECM hiệu chỉnh sai số.
\end{dinhnghia}
