\section{Cơ sở toán bảo hiểm}

\begin{dinhnghia}[title={Nguyên tắc phí bảo hiểm}]
  Phí thuần: $\pi = E(X)$, $X$: tổng bồi thường. Nguyên tắc an toàn:
  \begin{itemize}
    \item Giá trị kỳ vọng: $\pi = (1 + \theta)E(X)$, $\theta > 0$: hệ số an toàn.
    \item Phương sai: $\pi = E(X) + \alpha\text{Var}(X)$.
    \item Độ lệch chuẩn: $\pi = E(X) + \beta\sigma(X)$.
    \item Nguyên tắc hữu dụng: $u(w) = E[u(w + \pi - X)]$.
  \end{itemize}
\end{dinhnghia}

\begin{dinhnghia}[title={Mô hình tổn thất}]
  \textbf{Tần suất}: $N \sim \text{Poisson}(\lambda)$, $\text{NB}(r, p)$, \ldots

  \textbf{Mức độ}: $X_i \sim \text{Exp}(\mu)$, $\text{Gamma}$, $\text{Pareto}$, $\text{Lognormal}$, \ldots

  Tổng bồi thường: $S = \sum_{i=1}^{N}X_i$ (compound distribution).
\end{dinhnghia}

\begin{congthuc}[title={Compound Poisson}]
  $E(S) = \lambda E(X_1)$, $\text{Var}(S) = \lambda E(X_1^2)$, $M_S(t) = \exp[\lambda(M_{X_1}(t) - 1)]$.
\end{congthuc}

\section{Mô hình rủi ro cổ điển (Cramér--Lundberg)}

\begin{dinhnghia}[title={Quá trình thặng dư}]
  $U(t) = u + ct - S(t)$, $u$: vốn ban đầu, $c$: tốc độ thu phí, $S(t) = \sum_{i=1}^{N(t)}X_i$.
\end{dinhnghia}

\begin{dinhnghia}[title={Xác suất phá sản}]
  $\psi(u) = \Pr(\exists\, t > 0: U(t) < 0)$.
\end{dinhnghia}

\begin{dinhly}[title={Phương trình Lundberg}]
  Nếu tồn tại $R > 0$ (hệ số điều chỉnh): $\lambda + cR = \lambda M_X(R)$, thì:
  \[
    \psi(u) \le e^{-Ru} \quad \text{(chặn Lundberg)}.
  \]
\end{dinhly}

\begin{dinhly}[title={Công thức Pollaczek--Khinchine}]
  Khi $X_i \sim \text{Exp}(\beta)$: $\psi(u) = \dfrac{\lambda}{\beta c}\exp\!\left[-\left(\beta - \dfrac{\lambda}{c}\right)u\right]$.
\end{dinhly}

\section{Bảo hiểm nhân thọ}

\begin{dinhnghia}[title={Bảng tử vong}]
  $l_x$: số sống tại tuổi $x$. $d_x = l_x - l_{x+1}$. $q_x = d_x/l_x$: xác suất chết trong năm. $_tp_x = l_{x+t}/l_x$.
\end{dinhnghia}

\begin{congthuc}[title={Giá trị hiện tại bảo hiểm}]
  Bảo hiểm trọn đời: $A_x = \sum_{k=0}^{\infty}v^{k+1}\,{}_kp_x\,q_{x+k}$, $v = (1+i)^{-1}$.

  Niên kim trọn đời: $\ddot{a}_x = \sum_{k=0}^{\infty}v^k\,{}_kp_x$.

  Quan hệ: $A_x = 1 - d\,\ddot{a}_x$, $d = iv = i/(1+i)$.
\end{congthuc}

\begin{dinhnghia}[title={Dự phòng toán học}]
  $_tV_x = A_{x+t} - P_x\,\ddot{a}_{x+t}$, $P_x = A_x/\ddot{a}_x$: phí thuần hằng năm.
\end{dinhnghia}

\section{Tái bảo hiểm}

\begin{dinhnghia}[title={Các hình thức}]
  \begin{itemize}
    \item \textbf{Quota share}: chia tỷ lệ cố định $\alpha$.
    \item \textbf{Excess of loss}: bồi thường $\max(X - M, 0)$, $M$: mức giữ lại.
    \item \textbf{Stop-loss}: bồi thường $\max(S - d, 0)$.
  \end{itemize}
\end{dinhnghia}
