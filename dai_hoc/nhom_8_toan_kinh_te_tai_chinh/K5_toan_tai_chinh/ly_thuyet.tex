\section{Lãi suất và giá trị thời gian của tiền}

\begin{dinhnghia}[title={Lãi đơn và lãi kép}]
  Lãi đơn: $FV = PV(1 + rn)$. Lãi kép: $FV = PV(1 + r)^n$.

  Lãi kép liên tục: $FV = PVe^{rT}$.
\end{dinhnghia}

\begin{congthuc}[title={Annuity và Perpetuity}]
  Annuity: $PV = A\cdot\dfrac{1 - (1+r)^{-n}}{r}$. Perpetuity: $PV = \dfrac{A}{r}$.

  Growing perpetuity: $PV = \dfrac{A}{r - g}$ ($g < r$).
\end{congthuc}

\begin{dinhnghia}[title={Lãi suất hiệu dụng}]
  Ghép lãi $m$ lần/năm: $r_{\text{eff}} = \left(1 + \dfrac{r_{\text{nom}}}{m}\right)^m - 1$. Liên tục: $r_{\text{eff}} = e^{r_{\text{nom}}} - 1$.
\end{dinhnghia}

\section{Định giá trái phiếu}

\begin{dinhnghia}[title={Trái phiếu coupon}]
  Mệnh giá $F$, coupon $C$ mỗi kỳ, $n$ kỳ, lãi suất thị trường $y$:
  \[
    P = \sum_{t=1}^{n}\frac{C}{(1+y)^t} + \frac{F}{(1+y)^n} = C\cdot\frac{1-(1+y)^{-n}}{y} + F(1+y)^{-n}.
  \]
\end{dinhnghia}

\begin{dinhnghia}[title={Duration và Convexity}]
  \textbf{Macaulay duration}: $D = \frac{1}{P}\sum_{t=1}^{n}\frac{tC_t}{(1+y)^t}$.

  \textbf{Modified duration}: $D^* = D/(1+y)$. $\Delta P/P \approx -D^*\Delta y$.

  \textbf{Convexity}: $\text{Conv} = \frac{1}{P}\sum_{t=1}^{n}\frac{t(t+1)C_t}{(1+y)^{t+2}}$. Hiệu chỉnh bậc 2.
\end{dinhnghia}

\section{Lý thuyết danh mục đầu tư}

\begin{dinhnghia}[title={Suất sinh lợi và rủi ro}]
  $E(R_p) = \sum w_i E(R_i)$, $\sigma_p^2 = \sum_i\sum_j w_i w_j \sigma_{ij}$.
\end{dinhnghia}

\begin{dinhly}[title={Biên hiệu quả Markowitz}]
  $\min \mathbf{w}^T\boldsymbol{\Sigma}\mathbf{w}$ t.c. $\mathbf{w}^T\boldsymbol{\mu} = \mu_p$, $\mathbf{w}^T\mathbf{1} = 1$.

  Nghiệm: kết hợp hai danh mục biên (two-fund theorem).
\end{dinhly}

\begin{dinhly}[title={CAPM (Capital Asset Pricing Model)}]
  $E(R_i) = R_f + \beta_i[E(R_M) - R_f]$, $\beta_i = \dfrac{\text{Cov}(R_i, R_M)}{\text{Var}(R_M)}$.

  Đường thị trường chứng khoán (SML): giá tài sản hợp lý khi nằm trên SML.
\end{dinhly}

\section{Quyền chọn --- Cơ bản}

\begin{dinhnghia}[title={Call và Put}]
  \textbf{Call châu Âu}: quyền mua, payoff $\max(S_T - K, 0)$.

  \textbf{Put châu Âu}: quyền bán, payoff $\max(K - S_T, 0)$.
\end{dinhnghia}

\begin{dinhly}[title={Ngang giá Put--Call}]
  $C - P = S_0 - Ke^{-rT}$ (quyền chọn châu Âu).
\end{dinhly}

\begin{phuongphap}[title={Mô hình nhị thức (CRR)}]
  Một bước: $S$ lên $Su$ hoặc xuống $Sd$. Xác suất trung hòa rủi ro: $q = \dfrac{e^{r\Delta t} - d}{u - d}$.

  $C = e^{-r\Delta t}[qC_u + (1-q)C_d]$. Mở rộng $n$ bước: cây nhị thức.
\end{phuongphap}
