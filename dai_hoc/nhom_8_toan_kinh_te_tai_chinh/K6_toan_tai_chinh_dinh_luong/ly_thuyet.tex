\section{Chuyển động Brown và giải tích ngẫu nhiên}

\begin{dinhnghia}[title={Chuyển động Brown hình học (GBM)}]
  $dS_t = \mu S_t\,dt + \sigma S_t\,dW_t$. Nghiệm:
  \[
    S_t = S_0 \exp\!\left[\left(\mu - \tfrac{\sigma^2}{2}\right)t + \sigma W_t\right].
  \]
\end{dinhnghia}

\begin{dinhly}[title={Bổ đề Itô}]
  $f(t, S_t)$ với $dS = \mu S\,dt + \sigma S\,dW$:
  \[
    df = \left(\frac{\partial f}{\partial t} + \mu S\frac{\partial f}{\partial S} + \frac{1}{2}\sigma^2 S^2\frac{\partial^2 f}{\partial S^2}\right)dt + \sigma S\frac{\partial f}{\partial S}\,dW.
  \]
\end{dinhly}

\begin{dinhnghia}[title={Độ đo trung hòa rủi ro}]
  Dưới $\mathbb{Q}$: $dS_t = rS_t\,dt + \sigma S_t\,dW_t^{\mathbb{Q}}$.

  Giá hợp đồng: $V_0 = e^{-rT}\mathbb{E}^{\mathbb{Q}}[\text{payoff}(S_T)]$.
\end{dinhnghia}

\section{Mô hình Black--Scholes}

\begin{dinhly}[title={Công thức Black--Scholes}]
  Call châu Âu:
  \[
    C = S_0 N(d_1) - Ke^{-rT}N(d_2),
  \]
  Put: $P = Ke^{-rT}N(-d_2) - S_0 N(-d_1)$, với
  \[
    d_1 = \frac{\ln(S_0/K) + (r + \sigma^2/2)T}{\sigma\sqrt{T}}, \quad d_2 = d_1 - \sigma\sqrt{T}.
  \]
\end{dinhly}

\begin{dinhnghia}[title={PDE Black--Scholes}]
  $\dfrac{\partial V}{\partial t} + rS\dfrac{\partial V}{\partial S} + \dfrac{1}{2}\sigma^2 S^2\dfrac{\partial^2 V}{\partial S^2} = rV$.
\end{dinhnghia}

\section{Greeks}

\begin{congthuc}[title={Các chỉ số Greeks}]
  \begin{itemize}
    \item $\Delta = \dfrac{\partial V}{\partial S}$ (call: $N(d_1)$, put: $N(d_1) - 1$).
    \item $\Gamma = \dfrac{\partial^2 V}{\partial S^2} = \dfrac{N'(d_1)}{S\sigma\sqrt{T}}$.
    \item $\Theta = \dfrac{\partial V}{\partial t}$ (time decay).
    \item $\mathcal{V} = \dfrac{\partial V}{\partial \sigma} = S\sqrt{T}N'(d_1)$ (vega).
    \item $\rho = \dfrac{\partial V}{\partial r}$.
  \end{itemize}
\end{congthuc}

\begin{phuongphap}[title={Delta hedging}]
  Danh mục: $\Pi = V - \Delta S$. $d\Pi = (dV - \Delta\,dS) = r\Pi\,dt$ (phi rủi ro). Rebalance liên tục.
\end{phuongphap}

\section{Quản lý rủi ro}

\begin{dinhnghia}[title={Value at Risk (VaR)}]
  $\Pr(L > \text{VaR}_\alpha) = \alpha$. VaR tham số (chuẩn): $\text{VaR}_\alpha = -(\mu - z_\alpha\sigma)$.
\end{dinhnghia}

\begin{dinhnghia}[title={Expected Shortfall (CVaR)}]
  $\text{ES}_\alpha = E[L \mid L > \text{VaR}_\alpha]$. Coherent risk measure (VaR thì không).
\end{dinhnghia}

\begin{phuongphap}[title={Mô phỏng Monte Carlo}]
  Sinh $N$ đường $S_T^{(j)}$ từ GBM. Ước lượng: $\hat{V}_0 = e^{-rT}\frac{1}{N}\sum_{j=1}^{N}\text{payoff}(S_T^{(j)})$.

  Giảm phương sai: biến đối kháng, biến kiểm soát, importance sampling.
\end{phuongphap}
